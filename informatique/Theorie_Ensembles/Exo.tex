\section*{Énoncé}
Soient les ensembles :
\begin{center}
    $E = \{x, y\}$ \quad $F = \{1, 2, 3\}$ \quad $G = \{a, b\}$
\end{center}
Leurs cardinaux respectifs sont : $|E| = 2$, $|F| = 3$, $|G| = 2$.

\section*{Exercice 1}
\begin{align*}
    E \times F &= \{(x, 1), (x, 2), (x, 3), (y, 1), (y, 2), (y, 3)\} \\
    F \times E &= \{(1, x), (1, y), (2, x), (2, y), (3, x), (3, y)\} \\
    E \times G &= \{(x, a), (x, b), (y, a), (y, b)\} \\
    G \times E &= \{(a, x), (a, y), (b, x), (b, y)\} \\
    F \times G &= \{(1, a), (1, b), (2, a), (2, b), (3, a), (3, b)\} \\
    G \times F &= \{(a, 1), (a, 2), (a, 3), (b, 1), (b, 2), (b, 3)\}
\end{align*}

\section*{Exercice 2}
\begin{align*}
    E \times G \times F = \{ &(x, a, 1), (x, a, 2), (x, a, 3), (x, b, 1), (x, b, 2), (x, b, 3), \\
                             &(y, a, 1), (y, a, 2), (y, a, 3), (y, b, 1), (y, b, 2), (y, b, 3) \} \\[10pt]
    F \times G \times E = \{ &(1, a, x), (1, a, y), (1, b, x), (1, b, y), \\
                             &(2, a, x), (2, a, y), (2, b, x), (2, b, y), \\
                             &(3, a, x), (3, a, y), (3, b, x), (3, b, y) \}
\end{align*}

\section*{Exercice 3}
\begin{align*}
    E^3 &= \{ (x, x, x), (x, x, y), (x, y, x), (x, y, y), (y, x, x), (y, x, y), (y, y, x), (y, y, y) \} \\[10pt]
    G^3 &= \{ (a, a, a), (a, a, b), (a, b, a), (a, b, b), (b, a, a), (b, a, b), (b, b, a), (b, b, b) \} \\[10pt]
    F^3 &= \{ (1, 1, 1), (1, 1, 2), (1, 1, 3), (1, 2, 1), (1, 2, 2), (1, 2, 3), (1, 3, 1), (1, 3, 2), (1, 3, 3), \\
        &\phantom{=\{ \ } (2, 1, 1), (2, 1, 2), (2, 1, 3), (2, 2, 1), (2, 2, 2), (2, 2, 3), (2, 3, 1), (2, 3, 2), (2, 3, 3), \\
        &\phantom{=\{ \ } (3, 1, 1), (3, 1, 2), (3, 1, 3), (3, 2, 1), (3, 2, 2), (3, 2, 3), (3, 3, 1), (3, 3, 2), (3, 3, 3) \}
\end{align*}

\section*{Exercice 4}
\begin{itemize}
    \item $|E^2 \times F \times G| = |E|^2 \times |F| \times |G| = 2^2 \times 3 \times 2 = 4 \times 3 \times 2 = \mathbf{24}$
    \item $|E \times F^2 \times G^2| = |E| \times |F|^2 \times |G|^2 = 2 \times 3^2 \times 2^2 = 2 \times 9 \times 4 = \mathbf{72}$
\end{itemize}
