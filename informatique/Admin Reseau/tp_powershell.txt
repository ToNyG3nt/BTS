TP2 Script de partage :

    $path = "C:\share"

    if (Test-Path $path) {
        Write-Host "Le dossier existe déjà"
        exit
    } else {
        New-Item -Path $path -ItemType Directory
        New-SmbShare -Name "share" -Path $path -FullAccess "Administrateurs"
    }

TP3 Liste d'utilisateurs :

    $filePath = "users.txt"

    if (Test-Path $filePath) {
        Remove-Item $filePath
    }

    $users = Get-LocalUser
    $count = $users.Count
    $time = Get-Date -Format "HH:mm"

    "$time $count" | Out-File $filePath
    $users | Select-Object * | Out-File $filePath -Append

TP4 Réalisation d'un menu :

    do {
        Write-Host "1. Affichage répertoire courant"
        Write-Host "2. Liste des fichiers du répertoire"
        Write-Host "3. Afficher le nom de l'utilisateur courant"
        Write-Host "4. Afficher le nom de la machine"
        Write-Host "5. Quitter"

        $choice = Read-Host "Votre choix"

        switch ($choice) {
            1 { Get-Location }
            2 { Get-ChildItem }
            3 { $env:USERNAME }
            4 { $env:COMPUTERNAME }
            5 { break }
            default { Write-Host "Erreur de saisie" }
        }
    } while ($choice -ne 5)
