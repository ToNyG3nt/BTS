\section{Identification du Poste}
\begin{itemize}[label=\textbullet]
    \item \textbf{Département :} Direction Générale / Direction Technique (R\&D).
    \item \textbf{Hiérarchie :} Rapport direct au Conseil d'Administration.
    \item \textbf{Mission principale :} Garantir la souveraineté technologique de la solution anti-cheat et assurer la rentabilité économique de l'entreprise.
\end{itemize}

\section{Responsabilités Stratégiques (Pôle CEO)}
\begin{itemize}[label=\textbullet]
    \item \textbf{Gestion Business :} Définition des objectifs de croissance et pilotage du budget R\&D.
    \item \textbf{Reporting :} Présentation des résultats financiers aux actionnaires.
    \item \textbf{Conformité :} Respect strict du RGPD pour les méthodes de télémétrie et de scan mémoire.
\end{itemize}

\section{Expertise Technique (Pôle CTO)}
\begin{itemize}[label=\textbullet]
    \item \textbf{Architecture Système :} Conception de drivers Windows (Ring 0) et mécanismes d'auto-protection (Anti-debug, Anti-VM).
    \item \textbf{Sécurité :} Revue de code critique pour prévenir les écrans bleus (BSOD).
    \item \textbf{Arbitrage :} Choix des technologies (C++, Rust, Go) et supervision du cycle de détection des triches.
\end{itemize}

\section{Management et Évaluation}
\begin{itemize}[label=\textbullet]
    \item \textbf{RH :} Recrutement d'ingénieurs système et formation continue sur les vulnérabilités OS.
    \item \textbf{Indicateurs (KPIs) :} Taux de détection des cheats, stabilité logicielle et atteinte de la marge brute.
\end{itemize}
