\section{AGENT SUPPORT 24/7 (H/F)}

\subsection{Identification du Poste}
\begin{itemize}[label=\textbullet]
    \item \textbf{Département} : Support Client
    \item \textbf{Supérieur} : Directeur Général
    \item \textbf{Collaborations} : Équipe technique, Commercial
    \item \textbf{Rôle} : Assurer une assistance continue aux studios clients et aux joueurs, à toute heure
\end{itemize}

\subsection{Missions Principales}

\subsubsection{Assistance aux Studios Clients}
\begin{itemize}[label=\textbullet]
    \item Répondre aux demandes des studios concernant l'intégration et l'utilisation du produit.
    \item Aider à la résolution de problèmes techniques de premier niveau.
    \item Escalader les incidents complexes à l'équipe technique avec un compte-rendu clair.
\end{itemize}

\subsubsection{Gestion des Signalements Joueurs}
\begin{itemize}[label=\textbullet]
    \item Traiter les signalements de faux positifs (joueurs bannis par erreur).
    \item Répondre aux joueurs de façon claire et courtoise via les canaux officiels.
    \item Documenter les cas récurrents pour aider l'équipe technique à améliorer le produit.
\end{itemize}

\subsubsection{Surveillance et Alertes}
\begin{itemize}[label=\textbullet]
    \item Surveiller les alertes du système pendant les plages de nuit et week-end.
    \item Notifier immédiatement l'équipe technique en cas d'incident grave.
    \item Rédiger des rapports d'activité pour chaque prise de poste.
\end{itemize}

\subsection{Compétences Requises}
\begin{itemize}[label=\textbullet]
    \item \textbf{Technique} : Notions en informatique, capacité à lire des logs basiques, à l'aise avec les outils de ticketing.
    \item \textbf{Humaines} : Patience, sens du service, réactivité, disponibilité pour les horaires décalés.
\end{itemize}

\subsection{Indicateurs de Réussite}
\begin{itemize}[label=\textbullet]
    \item Temps de réponse aux demandes et taux de résolution au premier contact.
    \item Satisfaction des clients et qualité des remontées vers l'équipe technique.
\end{itemize}
