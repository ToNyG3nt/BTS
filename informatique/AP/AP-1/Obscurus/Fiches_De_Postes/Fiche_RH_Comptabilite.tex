\section{RESPONSABLE RH \& COMPTABILITÉ (H/F)}

\subsection{Identification du Poste}
\begin{itemize}[label=\textbullet]
    \item \textbf{Département} : Ressources Humaines / Finance
    \item \textbf{Supérieur} : Directeur Général
    \item \textbf{Collaborations} : Équipe technique, Responsable Sécurité
    \item \textbf{Rôle} : Recruter les bons profils et tenir les comptes de l'entreprise
\end{itemize}

\subsection{Missions Principales}

\subsubsection{Recrutement}
\begin{itemize}[label=\textbullet]
    \item Rechercher des profils techniques spécialisés (sécurité, développement système).
    \item Organiser des tests techniques en lien avec le CTO.
    \item Travailler l'image de l'entreprise pour attirer les bons candidats.
\end{itemize}

\subsubsection{Gestion du Personnel}
\begin{itemize}[label=\textbullet]
    \item Gérer les salaires, les congés et les formations.
    \item Veiller au bien-être des équipes, notamment en période de forte charge.
    \item S'assurer du respect du droit du travail et rédiger les contrats (dont clauses de confidentialité).
\end{itemize}

\subsubsection{Comptabilité}
\begin{itemize}[label=\textbullet]
    \item Enregistrer les entrées et sorties d'argent, gérer les factures.
    \item Émettre les factures clients et relancer les impayés.
    \item Préparer les déclarations fiscales et gérer la paie.
    \item Suivre la trésorerie et préparer les dossiers de subventions R\&D (CIR/CII).
\end{itemize}

\subsection{Compétences Requises}
\begin{itemize}[label=\textbullet]
    \item \textbf{Technique} : Recrutement, droit social, comptabilité, maîtrise d'Excel.
    \item \textbf{Humaines} : Rigueur, discrétion, sens du contact.
\end{itemize}

\subsection{Indicateurs de Réussite}
\begin{itemize}[label=\textbullet]
    \item Rapidité et qualité des recrutements, fidélisation des employés.
    \item Comptes tenus à jour et obligations fiscales respectées dans les délais.
\end{itemize}
