\section{COMMERCIAL \& CHARGÉ DE COMMUNICATION (H/F)}

\subsection{Identification du Poste}
\begin{itemize}[label=\textbullet]
    \item \textbf{Département} : Commercial / Communication
    \item \textbf{Supérieur} : Directeur Général
    \item \textbf{Collaborations} : Équipe technique, Responsable Sécurité
    \item \textbf{Rôle} : Trouver de nouveaux clients studios et représenter l'image de l'entreprise
\end{itemize}

\subsection{Missions Principales}

\subsubsection{Développement Commercial}
\begin{itemize}[label=\textbullet]
    \item Contacter et démarcher les studios de jeux vidéo en phase de développement.
    \item Comprendre les besoins de chaque studio pour leur proposer la solution adaptée.
    \item Présenter le produit, négocier les contrats et assurer le suivi après-vente.
    \item Proposer des modules supplémentaires selon les besoins (IA, protection réseau).
\end{itemize}

\subsubsection{Communication}
\begin{itemize}[label=\textbullet]
    \item Créer des supports de présentation et des études de cas clients.
    \item Représenter l'entreprise lors de salons professionnels (GDC, Gamescom).
    \item Gérer le site web et les réseaux sociaux de l'entreprise.
\end{itemize}

\subsubsection{Gestion de Crise}
\begin{itemize}[label=\textbullet]
    \item Surveiller les retours de la communauté en ligne (Reddit, Discord).
    \item Travailler avec le CTO pour préparer les réponses officielles en cas d'incident.
    \item Rédiger des communiqués clairs et adaptés selon la gravité du problème.
\end{itemize}

\subsection{Compétences Requises}
\begin{itemize}[label=\textbullet]
    \item \textbf{Commercial} : Expérience en vente, connaissance du secteur jeux vidéo, maîtrise d'un CRM.
    \item \textbf{Communication} : Bonne plume, anglais courant.
    \item \textbf{Humaines} : Diplomatie, réactivité, persévérance.
\end{itemize}

\subsection{Indicateurs de Réussite}
\begin{itemize}[label=\textbullet]
    \item Nombre de nouveaux contrats signés et taux de conversion des prospects.
    \item Image de l'entreprise et perception de la solution par la communauté.
\end{itemize}
