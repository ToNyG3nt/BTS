\section{DIRECTEUR GÉNÉRAL \& TECHNIQUE (H/F)}

\subsection{Identification du Poste}
\begin{itemize}[label=\textbullet]
    \item \textbf{Département} : Direction Générale / Technique
    \item \textbf{Supérieur} : Conseil d'Administration
    \item \textbf{Collaborations} : Équipes commerciale, juridique et technique
    \item \textbf{Rôle} : Diriger l'entreprise et assurer la qualité technique du produit anti-cheat
\end{itemize}

\subsection{Missions Principales}

\subsubsection{Gestion de l'Entreprise}
\begin{itemize}[label=\textbullet]
    \item Définir les objectifs et la stratégie de l'entreprise.
    \item Gérer le budget et valider les dépenses importantes.
    \item Faire des bilans réguliers auprès des investisseurs.
    \item S'assurer que les pratiques de l'entreprise respectent la loi (RGPD, éthique).
\end{itemize}

\subsubsection{Direction Technique}
\begin{itemize}[label=\textbullet]
    \item Superviser la conception et le développement du logiciel anti-cheat.
    \item Valider les choix technologiques (langages, outils, architecture).
    \item Suivre les mises à jour des détections, de la découverte d'un cheat à son blocage.
    \item Vérifier la stabilité du logiciel pour éviter tout problème chez les clients.
\end{itemize}

\subsubsection{Management}
\begin{itemize}[label=\textbullet]
    \item Recruter et évaluer les responsables techniques.
    \item Maintenir le niveau de compétences de l'équipe via des formations.
\end{itemize}

\subsection{Compétences Requises}
\begin{itemize}[label=\textbullet]
    \item \textbf{Technique} : Bonne connaissance des systèmes Windows, de la sécurité informatique et du développement bas-niveau.
    \item \textbf{Humaines} : Leadership, calme en situation de crise, vision long terme.
\end{itemize}

\subsection{Indicateurs de Réussite}
\begin{itemize}[label=\textbullet]
    \item Taux de détection des cheats et absence de bugs critiques.
    \item Atteinte des objectifs financiers de l'entreprise.
\end{itemize}
