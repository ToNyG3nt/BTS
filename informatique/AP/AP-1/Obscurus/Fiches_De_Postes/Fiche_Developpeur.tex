\section{DÉVELOPPEUR LOGICIEL (H/F)}

\subsection{Identification du Poste}
\begin{itemize}[label=\textbullet]
    \item \textbf{Département} : Technique / Développement
    \item \textbf{Supérieur} : Directeur Technique (CTO)
    \item \textbf{Collaborations} : Responsable Sécurité Offensive, équipe support
    \item \textbf{Rôle} : Développer et maintenir les composants du logiciel anti-cheat
\end{itemize}

\subsection{Missions Principales}

\subsubsection{Développement du Produit}
\begin{itemize}[label=\textbullet]
    \item Coder les modules de détection de triche (analyse mémoire, comportement réseau).
    \item Développer et maintenir le driver système qui surveille l'environnement des joueurs.
    \item Implémenter les nouvelles détections définies par le Responsable Sécurité.
    \item Participer aux revues de code et corriger les bugs signalés.
\end{itemize}

\subsubsection{Dashboard et Outils Internes}
\begin{itemize}[label=\textbullet]
    \item Développer et améliorer le tableau de bord utilisé par les studios clients.
    \item Créer des outils internes pour faciliter le travail de l'équipe sécurité.
\end{itemize}

\subsubsection{Qualité et Stabilité}
\begin{itemize}[label=\textbullet]
    \item Écrire des tests automatisés pour s'assurer que les mises à jour ne cassent rien.
    \item Optimiser les performances du logiciel pour réduire son impact sur les joueurs.
    \item Documenter le code pour faciliter la maintenance future.
\end{itemize}

\subsection{Compétences Requises}
\begin{itemize}[label=\textbullet]
    \item \textbf{Technique} : Maîtrise du C++ et/ou Rust, bases en développement Windows, notions de sécurité informatique appréciées.
    \item \textbf{Humaines} : Rigueur, autonomie, bonne communication avec l'équipe.
\end{itemize}

\subsection{Indicateurs de Réussite}
\begin{itemize}[label=\textbullet]
    \item Stabilité des livrables (peu de bugs en production).
    \item Respect des délais de développement des nouvelles fonctionnalités.
\end{itemize}
