\section{Exercice 1 : Calcul de la Factorielle}

\textbf{Énoncé :} Écrire un algorithme qui calcule la factorielle d'un nombre entier positif n.

\begin{algorithm}{Calcul de la factorielle (version itérative)}
\FUNCTION{Factorielle}{n} \\
\INDENT résultat $\leftarrow$ 1 \\
\INDENT \FOR{i de 1 à n} \\
\INDENT \INDENT résultat $\leftarrow$ résultat $\times$ i \\
\INDENT \ENDFOR \\
\INDENT \RETURN{résultat} \\
\KW{fin fonction}
\end{algorithm}
