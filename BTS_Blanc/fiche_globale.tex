\begin{tcolorbox}[basebox, colframe=teal, title=RÉSEAUX — Adressage IP]
\key{IPv4} : 32 bits, 4 octets (192.168.1.1) \\
\key{IPv6} : 128 bits, 8 groupes hex \\[2pt]
\textbf{Plages privées :}
\begin{itemize}
  \item 10.0.0.0/8
  \item 172.16.0.0/12
  \item 192.168.0.0/16
\end{itemize}
\key{CIDR} : /24 $\to$ 254 hôtes utiles \\
\key{Passerelle} = routeur par défaut \\
\key{DHCP} : attribution IP auto \\
\key{DNS} : nom $\leftrightarrow$ adresse IP
\end{tcolorbox}

% =============================================
\begin{tcolorbox}[basebox, colframe=navy, title=ROUTAGE \& VLAN]
\key{Routeur} : interconnecte réseaux (L3) \\
\key{Switch} : commute trames (L2) \\
\key{VLAN} : segmentation logique réseau \\
\quad $\to$ VLAN 10: Compta, VLAN 20: IT \\
\key{Trunk} : lien inter-switch multi-VLAN \\
\key{STP} : évite les boucles réseau \\[2pt]
\textbf{Cisco essentiels :}
\begin{itemize}
  \item \texttt{enable} / \texttt{conf t}
  \item \texttt{vlan 10} / \texttt{name IT}
  \item \texttt{show ip route}
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[basebox, colframe=crimson, title=CYBERSÉCURITÉ — Piliers]
\key{CIA} (Triade) :
\begin{itemize}
  \item \textbf{C}onfidentialité : accès autorisés
  \item \textbf{I}ntégrité : données non altérées
  \item \textbf{D}isponibilité : service accessible
\end{itemize}
\textbf{Menaces :}
\begin{itemize}
  \item \warn{Phishing} : usurpation e-mail
  \item \warn{Ransomware} : chiffrement+rançon
  \item \warn{XSS} : injection JS web
  \item \warn{SQLi} : injection SQL
\end{itemize}
\key{RBAC} : droits par rôle \\
\key{MFA} : authentification multi-facteur
\end{tcolorbox}

% =============================================
\begin{tcolorbox}[basebox, colframe=crimson, title=CYBERSÉCURITÉ — RGPD \& PCA]

\begin{itemize}
  \item Consentement explicite
  \item Droit à l'oubli, droit d'accès
  \item Notification breach : \warn{72h}
  \item Amende : jusqu'à \warn{4\% CA mondial}
\end{itemize}
\key{PCA} : maintenir l'activité \\
\key{PRA} : reprendre après sinistre \\
\key{RTO} : délai max de reprise \\
\key{RPO} : perte de données max acceptable
\end{tcolorbox}

\begin{tcolorbox}[basebox, colframe=navy, title=MATHS — Algèbre Booléenne]
{\footnotesize
\begin{tabular}{@{}ll@{}}
\key{ET (AND)} & $A \cdot B$ : vrai si A=1 et B=1 \\
\key{OU (OR)} & $A + B$ : vrai si A=1 ou B=1 \\
\key{NON (NOT)} & $\overline{A}$ : inverse de A \\
\key{XOR} & $A \oplus B$ : vrai si différents \\
\end{tabular}}
\vspace{2pt} \\
\textbf{Identités :} $A \cdot 0=0$ ; $A+1=1$ ; $\overline{\overline{A}}=A$ \\
\textbf{De Morgan :} $\overline{A \cdot B} = \overline{A}+\overline{B}$
\end{tcolorbox}
